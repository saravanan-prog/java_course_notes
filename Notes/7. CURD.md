## JDBC Flow Used in All Programs

- Load Driver
- Create Connection
- Create Statement / PreparedStatement
- Execute Query
- Process Result
- Close Connection





### 1. Read
```
import java.sql.*;

public class SelectExample {

    public static void main(String[] args) {

        try {

            Class.forName("com.mysql.cj.jdbc.Driver");

            Connection con =
                    DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/studentdb",
                    "root",
                    "password");

            Statement st =
                    con.createStatement();

            ResultSet rs =
                    st.executeQuery(
                    "SELECT * FROM students");

            while(rs.next()) {

                System.out.println(
                        rs.getInt("id") + " " +
                        rs.getString("name") + " " +
                        rs.getInt("marks")
                );
            }

            con.close();

        } catch(Exception e) {
            e.printStackTrace();
        }
    }
}

```

### 2. Add New Record
```
import java.sql.*;

public class InsertExample {

    public static void main(String[] args) {

        try {

            Class.forName("com.mysql.cj.jdbc.Driver");

            Connection con =
                    DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/studentdb",
                    "root",
                    "password");

            String query =
                    "INSERT INTO students VALUES (?, ?, ?)";

            PreparedStatement ps =
                    con.prepareStatement(query);

            ps.setInt(1, 101);
            ps.setString(2, "Saravanan");
            ps.setInt(3, 95);

            int rows = ps.executeUpdate();

            System.out.println(
                    rows + " Record Inserted"
            );

            con.close();

        } catch(Exception e) {
            e.printStackTrace();
        }
    }
}
```

###  3. UPDATE Example

```
import java.sql.*;

public class UpdateExample {

    public static void main(String[] args) {

        try {

            Class.forName("com.mysql.cj.jdbc.Driver");

            Connection con =
                    DriverManager.getConnection(
                    "jdbc:mysql://localhost:3306/studentdb",
                    "root",
                    "password");

                    "UPDATE students SET marks=? WHERE id=?";

            PreparedStatement ps =
                    con.prepareStatement(query);

            ps.setInt(2, 101);

                    ps.executeUpdate();

            System.out.println(

            con.close();

        } catch(Exception e) {
            e.printStackTrace();
        }
    }
}
```

### 4. DELETE Example
